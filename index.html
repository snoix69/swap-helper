async function loadNFTData() { try { // Запрос к API Getgems const query = `query { nft(address: "${nftAddress}") { name previews { url } owner { address } attributes { traitType value } } }`; const res = await fetch('https://api.getgems.io/graphql', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ query }) }); const json = await res.json(); const data = json.data.nft; if (!data) throw new Error("NFT not found"); document.getElementById('giftTitle').innerText = data.name || 'NFT Gift'; if (data.previews && data.previews.length > 0) { document.getElementById('giftImage').src = data.previews[0].url; } const owner = data.owner?.address || 'Unknown'; document.getElementById('ownerText').innerText = owner.substring(0, 10) + '...' + owner.substring(owner.length - 6); if (data.attributes) { data.attributes.forEach(attr => { const val = `${attr.value} ${attr.rarity || ''}%`; if (attr.traitType === 'Model') document.getElementById('modelText').innerHTML = val; if (attr.traitType === 'Backdrop') document.getElementById('backdropText').innerHTML = val; if (attr.traitType === 'Symbol') document.getElementById('symbolText').innerHTML = val; }); } } catch (e) { console.error("Ошибка загрузки данных:", e); document.getElementById('giftTitle').innerText = "Vintage Cigar #459"; document.getElementById('giftImage').src = "https://nft.fragment.com/gift/vintagecigar-459.webp"; document.getElementById('modelText').innerText = "Pink Ghost 3%"; document.getElementById('backdropText').innerText = "Caramel 2%"; document.getElementById('symbolText').innerText = "Genie Lamp 1%"; } }
